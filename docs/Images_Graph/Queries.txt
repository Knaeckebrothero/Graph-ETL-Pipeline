# Basic Stats

// Node and relationship overview
MATCH (n) RETURN labels(n)[0] AS Label, count(*) AS Count ORDER BY Count DESC

// Relationship type breakdown
MATCH ()-[r]->() RETURN type(r) AS Relationship, count(r) AS Count ORDER BY Count DESC

// Language distribution
MATCH (d:DictionaryEntry) RETURN d.lang AS Language, count(d) AS Count ORDER BY Count DESC

# Synonym Network

// Schokolade synonym cluster (visual graph)
MATCH (d:DictionaryEntry {name: "Schokolade"})<-[:SYNONYM_OF]-(s)
RETURN d, s

// Largest synonym clusters (top 10)
MATCH (canonical:DictionaryEntry)<-[:SYNONYM_OF]-(syn)
RETURN canonical.name AS Concept, count(syn) AS Synonyms
ORDER BY Synonyms DESC LIMIT 10

# Taxonomy (IS_A Hierarchy)

// Multi-level IS_A tree: Wasserfahrzeug → Fahrzeug → Verkehrswesen
MATCH path = (child:DictionaryEntry {name: "Aak"})-[:IS_A*1..3]->(ancestor)
RETURN path

// Top-level hypernyms (most children)
MATCH (child)-[:IS_A]->(parent:DictionaryEntry)
RETURN parent.name AS Hypernym, count(child) AS Children
ORDER BY Children DESC LIMIT 10

# Cross-Language Links

// German → English translation pairs (sample)
MATCH (en:DictionaryEntry {lang: "en"})-[:TRANSLATION_OF]->(de:DictionaryEntry {lang: "de"})
RETURN de.name AS German, en.name AS English
ORDER BY de.name LIMIT 20

# Combined Graph View (best for visual screenshots)

// Full neighborhood of a waste-relevant term
MATCH (d:DictionaryEntry {name: "Kleidung"})
OPTIONAL MATCH (d)<-[s:SYNONYM_OF]-(syn)
MATCH (d)-[i:IS_A]->(parent)
MATCH (en)-[t:TRANSLATION_OF]->(d)
RETURN *

// Provenance: entries derived from dictionary source
MATCH (d:DictionaryEntry)-[r:DERIVED_FROM]->(s:Source)
RETURN s, d, r LIMIT 25

# Waste-Relevant Concepts

// Waste-related terms in the dictionary
MATCH (d:DictionaryEntry)
WHERE d.name IN ["Abfall", "Müll", "Batterie", "Plastik", "Glas", "Papier", "Biomüll", "Schrott", "Elektroschrott"]
OPTIONAL MATCH (d)<-[:SYNONYM_OF]-(syn)
OPTIONAL MATCH (d)-[:IS_A]->(parent)
RETURN d, syn, parent
